\begin{enumerate}[label=\thesubsection.\arabic*, ref=\thesubsection.\theenumi]

\item Probability Mass Function (PMF) and Z-Transform of Discrete Random Variables:
\begin{align}
M_Y(Z) = E[Z^{-Y}] = \sum_k P_Y(k) Z^{-k}
\end{align}

\item Z-Transform of Sum of Independent Random Variables $Y = \sum_{i=1}^n Y_i$:
\begin{align}
M_Y(Z) = \prod_{i=1}^n M_{Y_i}(Z)
\end{align}

\item Combinations with Repetition (Stars and Bars Principle):
\begin{align}
N = \binom{n + r - 1}{r}
\end{align}

\item Composition of Functions and Surjectivity (Onto Property):
\begin{align}
(f \circ g)(x) = f(g(x)), \quad \forall x \in A
\end{align}

\item Shortest Path Inequality in Subgraphs and Spanning Trees:
If $T$ is a minimum spanning tree of graph $G$, then for any vertices $u, v \in V(G)$:
\begin{align}
d_1(u, v) \le d_2(u, v)
\end{align}
where $d_1(u,v)$ is the shortest path in $G$ and $d_2(u,v)$ is the shortest path in $T$.

\item Recurrence Relation Reduction using Exponential Division:
\begin{align}
T(n) = 2T(n-1) + n 2^n \implies \frac{T(n)}{2^n} = \frac{T(n-1)}{2^{n-1}} + n \implies T(n) = \Theta(n^2 2^n)
\end{align}

\item Determinant Conditions for Linear System Consistency $\mathbf{A}\mathbf{x} = \mathbf{b}$:
\begin{align}
\det(\mathbf{A}) \neq 0 \iff \text{Unique Solution}
\end{align}

\item Majority Boolean Function Representation:
\begin{align}
X(a, b, c) = ab + bc + ca
\end{align}

\item Sign Extension Rule in 2's Complement Numbers:
To extend an $n$-bit signed integer to an $m$-bit integer ($m > n$), replicate the Most Significant Bit (MSB / sign bit) across all added higher-order bit positions.

\item Number of Strings over Alphabet Size $k$ without Adjacent Duplicate Symbols:
\begin{align}
N_{\text{no\_consec}} = k(k-1)^{n-1}
\end{align}

\item Boundary Differentiability Condition for Piecewise Functions:
\begin{align}
\lim_{x \to x_0^-} f'(x) = \lim_{x \to x_0^+} f'(x)
\end{align}

\item Bayes' Theorem for Conditional Probability:
\begin{align}
P(A|B) = \frac{P(B|A)P(A)}{P(B)}
\end{align}

\item Total Inversions (Swaps) in a Reverse-Sorted Array of Size $n$:
\begin{align}
\text{Swaps} = \frac{n(n-1)}{2}
\end{align}

\item Eigenvalues of Matrix Powers:
If $\lambda$ is an eigenvalue of matrix $\mathbf{A}$, then $\lambda^k$ is an eigenvalue of $\mathbf{A}^k$:
\begin{align}
\mathbf{A}\mathbf{x} = \lambda \mathbf{x} \implies \mathbf{A}^k \mathbf{x} = \lambda^k \mathbf{x}
\end{align}

\item Minimum Height of Breadth-First-Search (BFS) Tree given Graph Diameter $D$:
\begin{align}
h(T) \ge \left\lceil \frac{D}{2} \right\rceil
\end{align}

\item Algebraic Monoid Structure under Function Addition:
A set of non-negative integer functions under pointwise addition $(f_1 \odot f_2)(n) = f_1(n) + f_2(n)$ forms an Abelian monoid with identity element $e(n) = 0$.

\item Basic Block Leader Criteria in Compiler Intermediate Code:
\begin{align}
\text{Leaders} = \{\text{First Instruction}, \text{Targets of Jumps}, \text{Instructions Immediately Following Jumps}\}
\end{align}

\item Independent Bernoulli Trials for $n$-bit Data Transmission:
\begin{align}
P(\text{Error-Free}) = (1 - p)^n
\end{align}
where $p$ is the bit-flip probability per bit.

\item Continuous Probability Density Function (PDF) Normalization:
\begin{align}
\int_{-\infty}^{\infty} P(x) \, dx = 1, \quad P(a \le x \le b) = \int_a^b P(x) \, dx
\end{align}

\item State Equivalence Reduction in Finite State Machines (FSM):
Two states $S_i$ and $S_j$ are equivalent if for all input symbols $x$:
\begin{align}
\text{Out}(S_i, x) = \text{Out}(S_j, x) \quad \text{and} \quad \text{Next}(S_i, x) \equiv \text{Next}(S_j, x)
\end{align}

\item Period of an $n$-bit Johnson / Twisted-Ring Counter:
An $n$-bit Johnson counter constructed with inverted feedback ($D_0 = \overline{Q_{n-1}}$) cycles through $2n$ distinct state transitions:
\begin{align}
N_{\text{states}} = 2n
\end{align}

\item Cycle Property of Minimum Spanning Trees (MST):
For any simple cycle $C$ in graph $G$, the strictly maximum-weight edge in $C$ cannot belong to any MST of $G$.

\item Open Addressing Double Hashing Probe Sequence:
\begin{align}
h(k, i) = \left[ h_1(k) + i \cdot h_2(k) \right] \bmod m
\end{align}
where $i \in \{0, 1, \dots, m-1\}$ represents the probe iteration count and $m$ is the hash table capacity.

\end{enumerate}
